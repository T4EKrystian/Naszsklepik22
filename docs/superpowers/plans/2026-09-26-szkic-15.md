- [ ] **Krok 1: `tools/split.py`**
```python
import re, sys, json, hashlib, base64, os
src, out_html, out_dir = sys.argv[1:4]
s = open(src, encoding='utf-8').read()
os.makedirs(out_dir, exist_ok=True)
EXT = {'image/jpeg':'jpg','image/png':'png','font/woff2':'woff2'}
man = {}
def rep(m):
    mime, b64 = m.group(1), m.group(2)
    aid = hashlib.sha1(b64.encode()).hexdigest()[:12]
    if aid not in man:
        man[aid] = mime
        open(f'{out_dir}/{aid}.{EXT[mime]}', 'wb').write(base64.b64decode(b64))
    return '{{a:%s}}' % aid
t = re.sub(r'data:(image/jpeg|image/png|font/woff2);base64,([A-Za-z0-9+/=]+)', rep, s)
open(out_html, 'w', encoding='utf-8').write(t)
json.dump(man, open(f'{out_dir}/manifest.json', 'w'), indent=0)
print(len(man), 'assets;', len(t), 'bytes source')
```
- [ ] **Krok 2: `tools/build.py`**. Wstawia `data:<mime>;base64,<plik>` w miejsce każdego `{{a:ID}}`. Nowe assety (np. z Higgsfield) dodaje się, wrzucając plik do `src/assets/` i wpis do `manifest.json` z identyfikatorem wybranym ręcznie, np. `hf_obs_wide`.
```python
import re, sys, json, base64
src, assets, out = sys.argv[1:4]
man = json.load(open(f'{assets}/manifest.json'))
EXT = {'image/jpeg':'jpg','image/png':'png','font/woff2':'woff2','image/webp':'webp'}
def rep(m):
    aid = m.group(1); mime = man[aid]
    return f'data:{mime};base64,' + base64.b64encode(open(f'{assets}/{aid}.{EXT[mime]}','rb').read()).decode()
html = re.sub(r'\{\{a:([A-Za-z0-9_]+)\}\}', rep, open(src, encoding='utf-8').read())
open(out, 'w', encoding='utf-8').write(html)
print(out, len(html)//1024, 'KB')
```
